## 二、数据导入脚本

### 2.1 导入角色数据
```bash
#!/bin/bash
# import_characters.sh

DB_HOST="127.0.0.1"
DB_USER="root"
DB_PASS="123456"
DB_NAME="heartsphere"
SQL_FILE="backend/src/main/resources/db/migration/add_daily_life_assistant_characters.sql"

echo "开始导入角色数据..."
mysql -h $DB_HOST -u $DB_USER -p$DB_PASS $DB_NAME --default-character-set=utf8mb4 < $SQL_FILE

if [ $? -eq 0 ]; then
  echo "✅ 角色数据导入成功"
  
  # 验证导入结果
  echo "验证导入结果..."
  mysql -h $DB_HOST -u $DB_USER -p$DB_PASS $DB_NAME -e "
    SELECT 
      '时代表' as type,
      COUNT(*) as count
    FROM system_eras WHERE name = '日常生活助手'
    UNION ALL
    SELECT 
      '角色' as type,
      COUNT(*) as count
    FROM system_characters 
    WHERE system_era_id = (SELECT id FROM system_eras WHERE name = '日常生活助手' LIMIT 1);
  "
else
  echo "❌ 角色数据导入失败"
  exit 1
fi
```

## 三、数据验证脚本

### 3.1 完整数据验证
```bash
#!/bin/bash
# verify_all_data.sh

DB_HOST="127.0.0.1"
DB_USER="root"
DB_PASS="123456"
DB_NAME="heartsphere"

echo "开始验证所有数据..."

mysql -h $DB_HOST -u $DB_USER -p$DB_PASS $DB_NAME <<EOF
SELECT 
  '时代表' as type,
  COUNT(*) as count,
  CASE WHEN COUNT(*) = 1 THEN '✅' ELSE '❌' END as status
FROM system_eras WHERE name = '日常生活助手'
UNION ALL
SELECT 
  '角色' as type,
  COUNT(*) as count,
  CASE WHEN COUNT(*) = 6 THEN '✅' ELSE '❌' END as status
FROM system_characters 
WHERE system_era_id = (SELECT id FROM system_eras WHERE name = '日常生活助手' LIMIT 1)
UNION ALL
SELECT 
  '技能定义' as type,
  COUNT(*) as count,
  CASE WHEN COUNT(*) = 48 THEN '✅' ELSE '❌' END as status
FROM skill_definitions
WHERE skill_id LIKE 'time_%' OR skill_id LIKE 'health_%' OR skill_id LIKE 'learn_%' 
   OR skill_id LIKE 'emotion_%' OR skill_id LIKE 'mental_%' OR skill_id LIKE 'companion_%'
UNION ALL
SELECT 
  '技能绑定' as type,
  COUNT(*) as count,
  CASE WHEN COUNT(*) = 48 THEN '✅' ELSE '❌' END as status
FROM character_skill_bindings csb
JOIN system_characters sc ON csb.character_id = sc.id
WHERE sc.system_era_id = (SELECT id FROM system_eras WHERE name = '日常生活助手' LIMIT 1);
EOF

echo "数据验证完成"
```

## 四、回滚脚本

### 4.1 回滚所有数据
```bash
#!/bin/bash
# rollback_all.sh

DB_HOST="127.0.0.1"
DB_USER="root"
DB_PASS="123456"
DB_NAME="heartsphere"

echo "⚠️  警告：即将删除所有集成数据！"
read -p "确认继续？(yes/no): " confirm

if [ "$confirm" != "yes" ]; then
  echo "已取消"
  exit 0
fi

echo "开始回滚数据..."

mysql -h $DB_HOST -u $DB_USER -p$DB_PASS $DB_NAME <<EOF
-- 删除技能绑定
DELETE FROM character_skill_bindings 
WHERE character_id IN (
  SELECT id FROM system_characters 
  WHERE system_era_id = (SELECT id FROM system_eras WHERE name = '日常生活助手' LIMIT 1)
);

-- 删除技能定义
DELETE FROM skill_definitions 
WHERE skill_id LIKE 'time_%' OR skill_id LIKE 'health_%' OR skill_id LIKE 'learn_%' 
   OR skill_id LIKE 'emotion_%' OR skill_id LIKE 'mental_%' OR skill_id LIKE 'companion_%';

-- 删除角色
DELETE FROM system_characters 
WHERE system_era_id = (SELECT id FROM system_eras WHERE name = '日常生活助手' LIMIT 1);

-- 删除时代表
DELETE FROM system_eras WHERE name = '日常生活助手';

SELECT '✅ 数据回滚完成' as result;
EOF

echo "✅ 回滚完成"
```
