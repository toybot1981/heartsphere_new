##### 1. multiCharacterDialogue类型
```typescript
// 修复前
const sortedDialogue = [...node.multiCharacterDialogue].sort((a: any, b: any) => ...);

// 修复后
interface DialogueItem {
  characterId: string;
  content: string;
  order?: number;
}
const sortedDialogue = [...(node.multiCharacterDialogue || [])].sort(
  (a: DialogueItem, b: DialogueItem) => (a.order || 0) - (b.order || 0)
);
```

##### 2. relevantMemories类型
```typescript
// 修复前
let relevantMemories: any[] = [];

// 修复后
interface MemoryItem {
  id: string;
  content: string;
  relevance?: number;
  [key: string]: unknown;
}
let relevantMemories: MemoryItem[] = [];
```

##### 3. 错误处理类型
```typescript
// 修复前
} catch (error: any) {
  showAlert('错误', error.message || '默认消息', 'error');
}

// 修复后
} catch (error: unknown) {
  const errorMessage = error instanceof Error ? error.message : '默认消息';
  showAlert('错误', errorMessage, 'error');
}
```

##### 4. webkitAudioContext类型
```typescript
// 修复前
audioPlayback.audioContextRef.current = new (window.AudioContext || (window as any).webkitAudioContext)({...});

// 修复后
type AudioContextType = typeof AudioContext;
const AudioContextClass = (window.AudioContext || (window as typeof window & { webkitAudioContext?: AudioContextType }).webkitAudioContext) as AudioContextType;
audioPlayback.audioContextRef.current = new AudioContextClass({...});
```

## 📝 类型定义最佳实践

### 1. 避免使用`any`
```typescript
// ❌ 不推荐
function processData(data: any) { ... }

// ✅ 推荐
interface DataItem {
  id: string;
  value: number;
}
function processData(data: DataItem) { ... }
```

### 2. 错误处理使用`unknown`
```typescript
// ❌ 不推荐
} catch (error: any) {
  console.error(error.message);
}

// ✅ 推荐
} catch (error: unknown) {
  const errorMessage = error instanceof Error ? error.message : '未知错误';
  console.error(errorMessage);
}
```

### 3. 使用类型联合
```typescript
// ✅ 推荐
type Status = 'pending' | 'loading' | 'success' | 'error';
function handleStatus(status: Status) { ... }
```

### 4. 使用泛型
```typescript
// ✅ 推荐
function processArray<T>(items: T[]): T[] {
  return items.map(item => transform(item));
}
```
